#include <windows.h>
#include <stdlib.h>
#include <stdio.h>
#include <time.h>
#include <math.h>

#ifdef __APPLE__
#include <GLUT/glut.h>
#else
#include <GL/glut.h>
#endif


// Basket Position
float basketX = 0.0f;
float basketWidth = 0.4f;

// Ball Position
float ballX = 0.0f;
float ballY = 1.0f;
float ballRadius = 0.05f;

int score = 0;

//------------------------------------
// Draw Text
//------------------------------------
void drawText(float x, float y, const char *text)
{
    glRasterPos2f(x, y);

    while (*text)
    {
        glutBitmapCharacter(GLUT_BITMAP_HELVETICA_18, *text);
        text++;
    }
}

//------------------------------------
// Draw Circle
//------------------------------------
void drawCircle(float x, float y, float r)
{
    glBegin(GL_POLYGON);

    for (int i = 0; i < 100; i++)
    {
        float angle = 2.0f * 3.1416f * i / 100;

        glVertex2f(
            x + r * cos(angle),
            y + r * sin(angle));
    }

    glEnd();
}

//------------------------------------
// Display
//------------------------------------
void display()
{
    glClear(GL_COLOR_BUFFER_BIT);

    // Basket
    glColor3f(0,0,1);

    glBegin(GL_QUADS);
        glVertex2f(basketX - basketWidth/2,-0.95);
        glVertex2f(basketX + basketWidth/2,-0.95);
        glVertex2f(basketX + basketWidth/2,-0.85);
        glVertex2f(basketX - basketWidth/2,-0.85);
    glEnd();

    // Ball
    glColor3f(1,0,0);
    drawCircle(ballX,ballY,ballRadius);

    // Score
    char str[50];
    sprintf(str,"Score : %d",score);

    glColor3f(1,1,1);
    drawText(-0.95,0.9,str);

    glutSwapBuffers();
}

//------------------------------------
// Animation
//------------------------------------
void update(int value)
{
    ballY -= 0.02f;

    // Collision
    if(ballY < -0.80 &&
       ballX > basketX-basketWidth/2 &&
       ballX < basketX+basketWidth/2)
    {
        score++;

        ballY = 1.0f;

        ballX = ((rand()%180)-90)/100.0f;
    }

    // Miss
    if(ballY < -1.1f)
    {
        ballY = 1.0f;

        ballX = ((rand()%180)-90)/100.0f;
    }

    glutPostRedisplay();

    glutTimerFunc(20,update,0);
}

//------------------------------------
// Keyboard
//------------------------------------
void specialKeys(int key,int x,int y)
{
    switch(key)
    {
    case GLUT_KEY_LEFT:
        basketX -= 0.08f;
        break;

    case GLUT_KEY_RIGHT:
        basketX += 0.08f;
        break;
    }

    if(basketX<-0.8)
        basketX=-0.8;

    if(basketX>0.8)
        basketX=0.8;

    glutPostRedisplay();
}

void keyboard(unsigned char key,int x,int y)
{
    if(key==27)
        exit(0);
}

//------------------------------------
// Initialization
//------------------------------------
void init()
{
    glClearColor(0,0,0,1);

    glMatrixMode(GL_PROJECTION);

    glLoadIdentity();

    gluOrtho2D(-1,1,-1,1);
}

//------------------------------------
// Main
//------------------------------------
int main(int argc,char** argv)
{
    srand(time(NULL));

    glutInit(&argc,argv);

    glutInitDisplayMode(GLUT_DOUBLE|GLUT_RGB);

    glutInitWindowSize(700,700);

    glutCreateWindow("Catch The Falling Ball");

    init();

    glutDisplayFunc(display);

    glutSpecialFunc(specialKeys);

    glutKeyboardFunc(keyboard);

    glutTimerFunc(20,update,0);

    glutMainLoop();

    return 0;
}
